2. AI Route Mocking Protocol (Zero Real API Calls)

Tests never invoke third-party LLM endpoints. All network transactions are mocked using global.fetch with realistic Server-Sent Events (SSE) ReadableStream payloads: 

// Pure client-side SSE stream mocking without real API costs
global.fetch = vi.fn().mockResolvedValue({
  ok: true,
  body: {
    getReader() {
      let step = 0;
      return {
        read() {
          const encoder = new TextEncoder();
          if (step === 0) {
            step++;
            return Promise.resolve({
              done: false,
              value: encoder.encode('data: {"type":"text-delta","text":"Hello World"}\n\n')
            });
          }
          return Promise.resolve({ done: true, value: undefined });
        }
      };
    }
  }
});
